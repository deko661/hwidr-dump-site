Decrypts/Common.hpp
#pragma once

#include <cstdint>
#include <cstddef>
#include <functional>

namespace Decrypt {

using ReadFn = std::function<bool(uint64_t addr, void* buf, size_t size)>;

inline uint32_t Rol32(uint32_t x, uint32_t n)
{
    n &= 31;
    return n ? (x << n) | (x >> (32 - n)) : x;
}

inline uint64_t Rol64(uint64_t x, uint32_t n)
{
    n &= 63;
    return n ? (x << n) | (x >> (64 - n)) : x;
}

inline uint8_t Rol8(uint8_t x, uint32_t n)
{
    n &= 7;
    return n ? (uint8_t)((x << n) | (x >> (8 - n))) : x;
}

template<typename T>
inline T Read(const ReadFn& rd, uint64_t addr)
{
    T v{};
    rd(addr, &v, sizeof(T));
    return v;
}

inline bool PtrOk(uint64_t p)
{
    p &= ~1ull;
    return p >= 0x10000 && p < 0x00007FFFFFFFFFFFull;
}

}

#pragma once

#include <string>
#include "Common.hpp"

namespace Decrypt::GNames {

constexpr uint32_t PoolRva    = 0x1B085580;
constexpr uint32_t BlocksOff  = 0x10;
constexpr uint32_t UObjectNameOff = 0x18;

inline uint32_t Bswap32(uint32_t x)
{
    return (x << 24) | ((x << 8) & 0x00FF0000u) | ((x >> 8) & 0x0000FF00u) | (x >> 24);
}

inline uint32_t DecodeIndex(uint32_t enc)
{
    if (!enc)
        return 0;
    const uint32_t t = Bswap32(enc - 1) + 0x2C11B315u;
    return t ? t : 0x2C11B314u;
}

inline uint32_t Seed(uint32_t len)
{
    const uint32_t hi = (len << 20) + len;
    const uint32_t m = len * 0x22CEu + 0xE2852u;
    return hi + (m ^ (m >> 11));
}

inline void DecryptBytes(const uint8_t* in, uint8_t* out, uint32_t len, uint32_t seed)
{
    uint32_t st = seed;
    for (uint32_t i = 0; i < len; ++i) {
        uint32_t mixed = st * 0x22CEu + 0xE2852u;
        mixed ^= mixed >> 11;
        st = (Rol32(st, 20) ^ st) + mixed;
        out[len - 1 - i] = (uint8_t)(in[i] + (uint8_t)st + 0x23u);
    }
}

inline std::string GetName(const ReadFn& rd, uint64_t base, uint32_t id, int depth = 0)
{
    if (depth > 8)
        return {};
    const uint64_t chunk = Read<uint64_t>(rd, base + PoolRva + BlocksOff + (uint64_t)(id >> 16) * 8);
    if (!PtrOk(chunk))
        return {};
    const uint64_t entry = chunk + (uint64_t)(id & 0xFFFF) * 2;
    const uint16_t header = Read<uint16_t>(rd, entry);
    const uint32_t len = (header & 0x3FF) ^ 0x266u;
    const bool wide = (header & 0x8000) != 0;
    if (!len) {
        const uint32_t redir = DecodeIndex(Read<uint32_t>(rd, entry + 6));
        if (!redir || redir == id)
            return {};
        return GetName(rd, base, redir, depth + 1);
    }
    if (len > 256)
        return {};
    const uint32_t bytes = wide ? len * 2 : len;
    uint8_t raw[512], dec[512];
    if (!rd(entry + 2, raw, bytes))
        return {};
    DecryptBytes(raw, dec, bytes, Seed(len));
    std::string s;
    if (wide) {
        for (uint32_t i = 0; i + 1 < bytes; i += 2) {
            const uint16_t c = (uint16_t)(dec[i] | (dec[i + 1] << 8));
            if (!c)
                break;
            s.push_back((c >= 32 && c < 127) ? (char)c : '?');
        }
    } else {
        for (uint32_t i = 0; i < len; ++i) {
            if (!dec[i])
                break;
            s.push_back((dec[i] >= 32 && dec[i] < 127) ? (char)dec[i] : '?');
        }
    }
    return s;
}

inline std::string GetObjectName(const ReadFn& rd, uint64_t base, uint64_t object)
{
    const uint32_t enc = Read<uint32_t>(rd, object + UObjectNameOff);
    return GetName(rd, base, DecodeIndex(enc));
}

}
Decrypts/GObjects.hpp
#pragma once

#include "Common.hpp"

namespace Decrypt::GObjects {

constexpr uint32_t ChunksRva   = 0x1B162F78;
constexpr uint32_t NumRva      = 0x1B162F80;
constexpr uint32_t ItemSize    = 0x18;
constexpr uint32_t ItemSerial  = 0xC;
constexpr uint32_t UObjectIndexOff = 0x14;

inline uint32_t Bswap32(uint32_t x)
{
    return (x << 24) | ((x << 8) & 0x00FF0000u) | ((x >> 8) & 0x0000FF00u) | (x >> 24);
}

inline uint64_t Bswap64(uint64_t x)
{
    return ((uint64_t)Bswap32((uint32_t)x) << 32) | Bswap32((uint32_t)(x >> 32));
}

inline uint64_t DecodeChunks(uint64_t enc)
{
    return Bswap64(enc) + 0x894E7C7Full;
}

inline int32_t DecodeNum(uint32_t enc)
{
    return (int32_t)(Bswap32(enc) + 0xEBB16CECu);
}

inline uint64_t DecodeItem(uint64_t packed, uint32_t enc32)
{
    if (packed & 0x1020000000000000ull)
        return 0;
    const uint32_t low = Bswap32(enc32) + 0x7765FD7Cu;
    return ((packed & 0x3FFF00000000ull) | low) << 3;
}

inline uint32_t DecodeObjectIndex(uint32_t enc)
{
    return Bswap32(enc) + 0x0887FA0Fu;
}

struct Table {
    uint64_t chunks = 0;
    int32_t num = 0;
};

inline Table Load(const ReadFn& rd, uint64_t base)
{
    Table t;
    t.chunks = DecodeChunks(Read<uint64_t>(rd, base + ChunksRva));
    t.num = DecodeNum(Read<uint32_t>(rd, base + NumRva));
    return t;
}

inline uint64_t GetObject(const ReadFn& rd, const Table& t, int32_t index)
{
    if (index < 0 || index >= t.num || !PtrOk(t.chunks))
        return 0;
    const uint64_t chunk = Read<uint64_t>(rd, t.chunks + (uint64_t)(index >> 16) * 8);
    if (!PtrOk(chunk))
        return 0;
    const uint64_t item = chunk + (uint64_t)(index & 0xFFFF) * ItemSize;
    const uint64_t packed = Read<uint64_t>(rd, item);
    const uint32_t enc32 = Read<uint32_t>(rd, item + 8);
    const uint64_t obj = DecodeItem(packed, enc32);
    if (!PtrOk(obj))
        return 0;
    if (DecodeObjectIndex(Read<uint32_t>(rd, obj + UObjectIndexOff)) != (uint32_t)index)
        return 0;
    return obj;
}

}

#pragma once

#include "Common.hpp"

namespace Decrypt::GWorld {

constexpr uint32_t Rva = 0x1B1D12F8;
constexpr uint32_t GetWorldVftIndexOff = 0x168;

inline uint64_t Bswap64(uint64_t x)
{
    const uint32_t lo = (uint32_t)x;
    const uint32_t hi = (uint32_t)(x >> 32);
    const uint32_t slo = (lo << 24) | ((lo << 8) & 0x00FF0000u) | ((lo >> 8) & 0x0000FF00u) | (lo >> 24);
    const uint32_t shi = (hi << 24) | ((hi << 8) & 0x00FF0000u) | ((hi >> 8) & 0x0000FF00u) | (hi >> 24);
    return ((uint64_t)slo << 32) | shi;
}

inline uint64_t Decode(uint64_t enc)
{
    const uint64_t w = Bswap64(enc) + 0xC731F903ull;
    return w ? w : 0;
}

inline uint64_t Get(const ReadFn& rd, uint64_t base)
{
    const uint64_t w = Decode(Read<uint64_t>(rd, base + Rva));
    return PtrOk(w) ? w : 0;
}

}
